from collections import deque

GRID = [
    ["S", ".", ".", ".", "."],
    ["X", ".", "X", ".", "."],
    [".", ".", ".", "X", "."],
    [".", "X", ".", ".", "."],
    [".", ".", ".", ".", "G"]
]

START = (0, 0)
GOAL = (4, 4)

DIRECTIONS = [(-1, 0), (1, 0), (0, -1), (0, 1)]


class ModelBasedAgent:
    def __init__(self, grid, start, goal):
        self.grid = grid
        self.start = start
        self.goal = goal
        self.rows = len(grid)
        self.cols = len(grid[0])
        self.model = [
            [None for _ in range(self.cols)]
            for _ in range(self.rows)
        ]
        self.position = start
        self.model[start[0]][start[1]] = "S"
        self.model[goal[0]][goal[1]] = "G"

    def perceive(self, position):
        r, c = position
        self.model[r][c] = self.grid[r][c]

    def plan_path(self):
        queue = deque([(self.position, [self.position])])
        visited = {self.position}

        while queue:
            (r, c), path = queue.popleft()

            if (r, c) == self.goal:
                return path

            for dr, dc in DIRECTIONS:
                nr, nc = r + dr, c + dc

                if not (0 <= nr < self.rows and
                        0 <= nc < self.cols):
                    continue

                if (nr, nc) in visited:
                    continue

                if self.model[nr][nc] == "X":
                    continue

                visited.add((nr, nc))
                queue.append(((nr, nc), path + [(nr, nc)]))

        return None

    def run(self):
        print("Grid:")
        for row in self.grid:
            print(row)

        self.perceive(self.position)
        print("\nAgent Navigation:")

        while self.position != self.goal:
            print("Current Position:", self.position)

            path = self.plan_path()

            if path is None or len(path) < 2:
                print("No path found!")
                return

            next_position = path[1]
            self.perceive(next_position)

            r, c = next_position

            if self.model[r][c] == "X":
                print("Obstacle discovered. Replanning...")
                continue

            self.position = next_position

        print("Current Position:", self.position)
        print("Goal Reached!")


agent = ModelBasedAgent(GRID, START, GOAL)
agent.run()
