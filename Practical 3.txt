
import heapq

GOAL = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)


def manhattan(state):
    distance = 0

    for index, tile in enumerate(state):
        if tile == 0:
            continue

        goal_index = GOAL.index(tile)
        row, col = divmod(index, 3)
        goal_row, goal_col = divmod(goal_index, 3)

        distance += abs(row - goal_row) + abs(col - goal_col)

    return distance


def is_solvable(state):
    values = [x for x in state if x != 0]

    inversions = sum(
        values[i] > values[j]
        for i in range(len(values))
        for j in range(i + 1, len(values))
    )

    return inversions % 2 == 0


def neighbors(state):
    zero = state.index(0)
    r, c = divmod(zero, 3)

    for dr, dc, move in [
        (-1, 0, "Up"),
        (1, 0, "Down"),
        (0, -1, "Left"),
        (0, 1, "Right")
    ]:
        nr, nc = r + dr, c + dc

        if 0 <= nr < 3 and 0 <= nc < 3:
            j = nr * 3 + nc
            new_state = list(state)
            new_state[zero], new_state[j] = (
                new_state[j], new_state[zero]
            )

            yield tuple(new_state), move


def astar(start):
    frontier = [(manhattan(start), 0, start)]
    parent = {start: (None, None)}
    best_g = {start: 0}

    while frontier:
        _, g, state = heapq.heappop(frontier)

        if g != best_g.get(state):
            continue

        if state == GOAL:
            moves = []
            states = []
            current = state

            while current is not None:
                states.append(current)
                previous, move = parent[current]

                if move is not None:
                    moves.append(move)

                current = previous

            return list(reversed(moves)), list(reversed(states))

        for nxt, move in neighbors(state):
            new_g = g + 1

            if new_g < best_g.get(nxt, float("inf")):
                best_g[nxt] = new_g
                parent[nxt] = (state, move)

                heapq.heappush(
                    frontier,
                    (new_g + manhattan(nxt), new_g, nxt)
                )

    return None, None


def print_state(state):
    for i in range(0, 9, 3):
        print(" ".join(
            "_" if x == 0 else str(x)
            for x in state[i:i + 3]
        ))
    print()


def main():
    try:
        numbers = [
            int(x)
            for x in input(
                "Enter 9 numbers (0 represents blank): "
            ).split()
        ]
    except ValueError:
        print("Please enter integers only.")
        return

    if len(numbers) != 9 or sorted(numbers) != list(range(9)):
        print("Invalid puzzle: enter each number from 0 to 8 exactly once.")
        return

    start = tuple(numbers)

    if not is_solvable(start):
        print("This puzzle is not solvable.")
        return

    moves, states = astar(start)

    if moves is None:
        print("No solution found.")
        return

    print("\nInitial state:")
    print_state(start)

    print("Moves:", moves)
    print("Number of moves:", len(moves))

    print("\nSolution states:")

    for step, state in enumerate(states):
        print(f"Step {step}:")
        print_state(state)


if __name__ == "__main__":
    main()
