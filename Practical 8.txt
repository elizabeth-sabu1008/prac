
WIN_LINES = [
    (0, 1, 2), (3, 4, 5), (6, 7, 8),
    (0, 3, 6), (1, 4, 7), (2, 5, 8),
    (0, 4, 8), (2, 4, 6)
]


def winner(board):
    for a, b, c in WIN_LINES:
        if board[a] != " " and board[a] == board[b] == board[c]:
            return board[a]
    return None


def minimax(board, is_maximizing, depth=0):
    result = winner(board)

    if result == "O":
        return 10 - depth
    if result == "X":
        return depth - 10
    if " " not in board:
        return 0

    if is_maximizing:
        best_score = float("-inf")

        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                score = minimax(board, False, depth + 1)
                board[i] = " "
                best_score = max(best_score, score)

        return best_score

    best_score = float("inf")

    for i in range(9):
        if board[i] == " ":
            board[i] = "X"
            score = minimax(board, True, depth + 1)
            board[i] = " "
            best_score = min(best_score, score)

    return best_score


def best_move(board):
    best_score = float("-inf")
    move = None

    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(board, False, 1)
            board[i] = " "

            if score > best_score:
                best_score = score
                move = i

    return move


def print_board(board):
    for row in range(3):
        print(" " + " | ".join(board[3 * row:3 * row + 3]))

        if row < 2:
            print("---+---+---")


def main():
    board = [" "] * 9

    print("=== Tic-Tac-Toe using Minimax ===")
    print("You are X; computer is O. Choose cells 1-9:")
    print("1 | 2 | 3\n---+---+---\n4 | 5 | 6\n---+---+---\n7 | 8 | 9")

    while True:
        while True:
            try:
                cell = int(input("\nYour move (1-9): ")) - 1

                if 0 <= cell < 9 and board[cell] == " ":
                    break

                print("Invalid or occupied cell. Try again.")

            except ValueError:
                print("Enter a number from 1 to 9.")

        board[cell] = "X"
        print_board(board)

        if winner(board) or " " not in board:
            break

        move = best_move(board)

        if move is None:
            break

        board[move] = "O"
        print(f"\nComputer plays {move + 1}:")
        print_board(board)

        if winner(board) or " " not in board:
            break

    result = winner(board)

    if result == "X":
        print("Result: You win!")
    elif result == "O":
        print("Result: Computer wins!")
    else:
        print("Result: Draw!")


if __name__ == "__main__":
    main()
