from collections import deque

n = int(input("Enter number of disks: "))

start = (tuple(range(n, 0, -1)), (), ())
goal = ((), (), tuple(range(n, 0, -1)))

queue = deque([(start, [])])
visited = {start}

while queue:
    state, path = queue.popleft()

    if state == goal:
        print("\nSolution:")
        for move in path:
            print(move)
        print("Total moves:", len(path))
        break

    for i in range(3):
        for j in range(3):
            if i != j and state[i]:
                disk = state[i][-1]

                if not state[j] or disk < state[j][-1]:
                    new_state = [list(peg) for peg in state]
                    new_state[i].pop()
                    new_state[j].append(disk)
                    new_state = tuple(tuple(peg) for peg in new_state)

                    if new_state not in visited:
                        visited.add(new_state)
                        queue.append((
                            new_state,
                            path + [
                                f"Move disk {disk} from peg {i+1} to peg {j+1}"
                            ]
                        ))
